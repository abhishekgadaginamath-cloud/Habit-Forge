## STEP 9 — Sync Clerk users to MongoDB via webhook

Create server/routes/webhooks.js:

import { Webhook } from 'svix'
import User from '../models/User.js'

export default function webhookRoutes(app) {
  app.post('/api/webhooks/clerk',
    express.raw({ type: 'application/json' }),
    async (req, res) => {
      const wh = new Webhook(process.env.CLERK_WEBHOOK_SECRET)
      let evt
      try {
        evt = wh.verify(req.body, {
          'svix-id': req.headers['svix-id'],
          'svix-timestamp': req.headers['svix-timestamp'],
          'svix-signature': req.headers['svix-signature'],
        })
      } catch {
        return res.status(400).json({ error: 'Invalid webhook' })
      }

      if (evt.type === 'user.created') {
        await User.create({
          clerkUserId: evt.data.id,
          email: evt.data.email_addresses[0]?.email_address,
          username: evt.data.username || evt.data.first_name || 'Adventurer',
          xp: 0,
          level: 1,
          badges: [],
          isPremium: false,
        })
      }

      if (evt.type === 'user.deleted') {
        await User.deleteOne({ clerkUserId: evt.data.id })
      }

      res.json({ received: true })
    }
  )
}

Register it in server/index.js:
import webhookRoutes from './routes/webhooks.js'
webhookRoutes(app)

## STEP 12 — Send Clerk token from frontend API calls

In client/src/utils/api.js (or wherever Axios is configured):

import { useAuth } from '@clerk/clerk-react'

Create a custom hook for authenticated requests:
export function useApi() {
  const { getToken } = useAuth()

  const authFetch = async (url, options = {}) => {
    const token = await getToken()
    return fetch(url, {
      ...options,
      headers: {
        'Content-Type': 'application/json',
        Authorization: `Bearer ${token}`,
        ...options.headers,
      },
    })
  }

  return { authFetch }
}

Use this hook in all components that call the backend instead of plain fetch/axios.
